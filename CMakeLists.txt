cmake_minimum_required(VERSION 3.20)
project(egemma_native LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0")
if(NOT APPLE)
  message(FATAL_ERROR "The native engine requires macOS and Metal")
endif()
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
  OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/kernels.metal" EGEMMA_METAL_SOURCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/kernels.metal)
configure_file(native/kernels_source.h.in generated/kernels_source.h @ONLY)
option(EGEMMA_ENABLE_MPP "Build the optional Metal Performance Primitives matmul path (selected at runtime)" ON)
add_library(egemma_engine STATIC native/engine.mm)
if(EGEMMA_ENABLE_MPP)
  target_compile_definitions(egemma_engine PRIVATE EGEMMA_ENABLE_MPP=1)
endif()
set_target_properties(egemma_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(egemma_engine PUBLIC native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(egemma_engine PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(egemma_engine PUBLIC "-framework Foundation" "-framework Metal" "-framework Accelerate")
add_executable(egemma-cli native/cli.mm)
target_compile_options(egemma-cli PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(egemma-cli PRIVATE egemma_engine)
pybind11_add_module(_egemma_native native/bindings.cpp)
target_compile_options(_egemma_native PRIVATE -Wall -Wextra -Werror)
target_link_libraries(_egemma_native PRIVATE egemma_engine)
